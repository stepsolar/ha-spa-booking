<script>
(function () {
  "use strict";
  // Regole per WordPress: niente righe vuote e nessuna "e commerciale" in
  // questo blocco, perché l'editor le trasforma e rompe lo script.
  var API_BASE = "https://hailmarchese.it/api/spa_booking/public";
  var TOKEN = "INCOLLA_QUI_IL_TOKEN";
  var MAX_DAYS_AHEAD = 60;
  var MAX_PEOPLE = 6;
  var AMP = String.fromCharCode(38);
  var root = document.getElementById("sbw-widget");
  if (!root) return;
  if (typeof fetch !== "function") return;
  function $(id) { return root.querySelector("#" + id); }
  var dateInput = $("sbw-date");
  var peopleSelect = $("sbw-people");
  var personsBox = $("sbw-persons");
  var voucherInput = $("sbw-voucher");
  var voucherBtn = $("sbw-voucher-check");
  var voucherInfo = $("sbw-voucher-info");
  var first = $("sbw-first");
  var last = $("sbw-last");
  var phone = $("sbw-phone");
  var email = $("sbw-email");
  var notes = $("sbw-notes");
  var sendBtn = $("sbw-send");
  var statusBox = $("sbw-status");
  var honey = $("sbw-website");
  var treatments = [];
  var voucherData = null;
  var sending = false;
  function el(tag, cls, text) {
    var node = document.createElement(tag);
    if (cls) node.className = cls;
    if (text !== undefined) node.textContent = text;
    return node;
  }
  function option(value, text, selected) {
    var o = el("option", "", text);
    o.value = value;
    if (selected) o.selected = true;
    return o;
  }
  function say(text, kind) {
    if (!statusBox) return;
    statusBox.textContent = text || "";
    statusBox.className = "sbw-status" + (kind ? " " + kind : "");
  }
  function pad2(n) { return (n < 10 ? "0" : "") + n; }
  function isoDay(d) { return d.getFullYear() + "-" + pad2(d.getMonth() + 1) + "-" + pad2(d.getDate()); }
  function describe(err) {
    var status = err ? err.status : 0;
    if (status === 404) return "Il servizio di prenotazione online non e attivo in questo momento: chiama la reception per prenotare.";
    if (status === 401) return "Il modulo non e configurato correttamente. Avvisa la reception.";
    if (status === 429) return "Troppe richieste in poco tempo: riprova tra qualche minuto.";
    if (status) { if (err.message) return err.message; }
    return "Non riesco a contattare il servizio di prenotazione: controlla la connessione e riprova, o chiama la reception.";
  }
  function api(path, options) {
    options = options || {};
    var headers = { "Authorization": "Bearer " + TOKEN };
    if (options.body) headers["Content-Type"] = "application/json";
    return fetch(API_BASE + path, {
      method: options.method || "GET",
      headers: headers,
      body: options.body ? JSON.stringify(options.body) : undefined,
      mode: "cors",
      cache: "no-store"
    }).then(function (r) {
      return r.text().then(function (raw) {
        var data = null;
        try { data = raw ? JSON.parse(raw) : null; } catch (e) { data = null; }
        if (!r.ok) {
          var failure = new Error((data ? data.message : null) || ("HTTP " + r.status));
          failure.status = r.status;
          throw failure;
        }
        return data;
      });
    }, function () {
      var offline = new Error("");
      offline.status = 0;
      throw offline;
    });
  }
  function treatmentById(id) {
    for (var i = 0; i < treatments.length; i++) { if (treatments[i].id === id) return treatments[i]; }
    return null;
  }
  function treatmentLabel(t) { return t.name + " (" + t.duration_min + " min)"; }
  function fillTreatments(select, selectedId, locked) {
    select.innerHTML = "";
    if (locked) {
      var t = treatmentById(selectedId);
      select.appendChild(option(selectedId, t ? treatmentLabel(t) : "Trattamento del voucher", true));
      select.disabled = true;
      return;
    }
    select.appendChild(option("", "-- scegli il trattamento --", !selectedId));
    treatments.forEach(function (t) { select.appendChild(option(t.id, treatmentLabel(t), t.id === selectedId)); });
  }
  function fillSlots(select, treatmentId) {
    var previous = select.value;
    select.innerHTML = "";
    if (!treatmentId) { select.appendChild(option("", "-- orario --", true)); return; }
    if (!dateInput.value) { select.appendChild(option("", "Scegli prima la data", true)); return; }
    select.appendChild(option("", "Caricamento...", true));
    var query = "/availability?date=" + encodeURIComponent(dateInput.value) + AMP + "treatment_id=" + encodeURIComponent(treatmentId);
    api(query).then(function (data) {
      var slots = (data ? data.slots : null) || [];
      select.innerHTML = "";
      if (!slots.length) { select.appendChild(option("", "Nessun orario libero", true)); return; }
      select.appendChild(option("", "-- orario --", false));
      slots.forEach(function (slot) { select.appendChild(option(slot.time, slot.time, slot.time === previous)); });
    }).catch(function (err) {
      select.innerHTML = "";
      select.appendChild(option("", "Orari non disponibili", true));
      say(describe(err), "err");
    });
  }
  function createEntryRow(treatmentId, locked) {
    var row = el("div", "sbw-entry");
    var treatmentSelect = el("select");
    var timeSelect = el("select", "sbw-time");
    treatmentSelect.setAttribute("aria-label", "Trattamento");
    timeSelect.setAttribute("aria-label", "Orario");
    fillTreatments(treatmentSelect, treatmentId || "", locked);
    treatmentSelect.addEventListener("change", function () { fillSlots(timeSelect, treatmentSelect.value); });
    row.appendChild(treatmentSelect);
    row.appendChild(timeSelect);
    fillSlots(timeSelect, treatmentId || "");
    return row;
  }
  function createPerson(index, personData) {
    var box = el("div", "sbw-person");
    var title = "Persona " + index;
    if (personData) { if (personData.name) title += " - " + personData.name; }
    box.appendChild(el("h4", "", title));
    var entries = el("div", "sbw-person-entries");
    box.appendChild(entries);
    var seeds = [];
    if (personData) { if (Array.isArray(personData.treatment_ids)) seeds = personData.treatment_ids; }
    if (seeds.length) {
      seeds.forEach(function (id) { entries.appendChild(createEntryRow(id, true)); });
    } else {
      entries.appendChild(createEntryRow("", false));
    }
    var addBtn = el("button", "sbw-btn sbw-small", "+ trattamento");
    addBtn.type = "button";
    addBtn.addEventListener("click", function () { entries.appendChild(createEntryRow("", false)); });
    box.appendChild(addBtn);
    return box;
  }
  function renderPersons() {
    if (!personsBox) return;
    var n = parseInt(peopleSelect.value || "1", 10);
    if (isNaN(n) || n < 1) n = 1;
    if (n > MAX_PEOPLE) n = MAX_PEOPLE;
    var people = voucherData ? (voucherData.people || []) : [];
    personsBox.innerHTML = "";
    for (var i = 0; i < n; i++) { personsBox.appendChild(createPerson(i + 1, people[i] || null)); }
  }
  function refreshAllSlots() {
    var rows = personsBox.querySelectorAll(".sbw-entry");
    for (var i = 0; i < rows.length; i++) {
      var selects = rows[i].querySelectorAll("select");
      if (selects.length > 1) fillSlots(selects[1], selects[0].value);
    }
  }
  function collectEntries() {
    // One entry per chosen treatment, with the person it is for (0 = the
    // first): confirming the request books exactly these, nothing doubled.
    var all = [];
    var persons = personsBox.querySelectorAll(".sbw-person");
    for (var p = 0; p < persons.length; p++) {
      var rows = persons[p].querySelectorAll(".sbw-entry");
      for (var i = 0; i < rows.length; i++) {
        var selects = rows[i].querySelectorAll("select");
        var treatment = selects[0] ? selects[0].value : "";
        var time = selects[1] ? selects[1].value : "";
        if (treatment) { if (time) all.push({ treatment_id: treatment, time: time, person: p }); }
      }
    }
    return all;
  }
  function loadTreatments() {
    return api("/treatments").then(function (data) {
      treatments = (data ? data.treatments : null) || [];
      renderPersons();
    }).catch(function (err) { say(describe(err), "err"); });
  }
  function loadVoucher() {
    var code = voucherInput.value.trim().toUpperCase();
    voucherData = null;
    if (!code) { voucherInfo.textContent = ""; renderPersons(); return; }
    voucherInfo.textContent = "Verifica in corso...";
    voucherBtn.disabled = true;
    api("/treatments?code=" + encodeURIComponent(code)).then(function (data) {
      voucherBtn.disabled = false;
      var voucher = data ? data.voucher : null;
      if (!voucher) { voucherInfo.textContent = "Voucher non valido."; renderPersons(); return; }
      voucherData = voucher;
      var people = voucher.people || [];
      var count = people.length;
      if (count < 1) count = 1;
      if (count > MAX_PEOPLE) count = MAX_PEOPLE;
      peopleSelect.value = String(count);
      var recipient = String(voucher.recipient_name || "").trim();
      if (recipient) {
        if (!first.value.trim()) {
          if (!last.value.trim()) {
            var parts = recipient.split(" ");
            first.value = parts.shift();
            last.value = parts.join(" ");
          }
        }
      }
      voucherInfo.textContent = "Voucher valido: " + code + " (" + people.length + " persona/e). I trattamenti del voucher sono gia selezionati: scegli solo gli orari.";
      renderPersons();
    }).catch(function (err) {
      voucherBtn.disabled = false;
      voucherInfo.textContent = err.status ? (err.message || "Voucher non valido.") : describe(err);
      renderPersons();
    });
  }
  function showReceipt(data) {
    var form = $("sbw-form");
    var done = $("sbw-done");
    if (form) form.hidden = true;
    if (!done) return;
    done.hidden = false;
    done.innerHTML = "";
    var receipt = el("div", "sbw-receipt");
    receipt.appendChild(el("div", "sbw-ok", "✓"));
    receipt.appendChild(el("p", "", "Richiesta inviata! La reception ti contattera per confermare l'appuntamento."));
    receipt.appendChild(el("p", "", "Codice della richiesta:"));
    receipt.appendChild(el("div", "sbw-code", (data ? data.code : "") || "—"));
    done.appendChild(receipt);
  }
  function submit() {
    if (sending) return;
    if (honey) { if (honey.value) return; }
    if (!dateInput.value) { say("Scegli la data.", "err"); return; }
    var entries = collectEntries();
    if (!entries.length) { say("Scegli almeno un trattamento con il suo orario.", "err"); return; }
    var firstVal = first.value.trim();
    var lastVal = last.value.trim();
    var phoneVal = phone.value.trim();
    if (!firstVal) { if (!lastVal) { say("Indica il tuo nome.", "err"); return; } }
    if (!phoneVal) { say("Indica un numero di telefono: serve per confermare.", "err"); return; }
    sending = true;
    sendBtn.disabled = true;
    say("Invio la richiesta...");
    api("/requests", { method: "POST", body: {
      date: dateInput.value,
      entries: entries,
      first_name: firstVal,
      last_name: lastVal,
      phone: phoneVal,
      email: email.value.trim(),
      notes: notes.value.trim(),
      voucher_code: voucherData ? voucherData.code : voucherInput.value.trim().toUpperCase(),
      people: parseInt(peopleSelect.value, 10) || 1
    } }).then(function (data) {
      sending = false;
      sendBtn.disabled = false;
      say("", "ok");
      showReceipt(data);
    }).catch(function (err) {
      sending = false;
      sendBtn.disabled = false;
      say(describe(err), "err");
      refreshAllSlots();
    });
  }
  var today = new Date();
  today.setHours(12, 0, 0, 0);
  dateInput.min = isoDay(today);
  dateInput.max = isoDay(new Date(today.getTime() + MAX_DAYS_AHEAD * 86400000));
  dateInput.addEventListener("change", refreshAllSlots);
  peopleSelect.addEventListener("change", renderPersons);
  voucherBtn.addEventListener("click", loadVoucher);
  voucherInput.addEventListener("keydown", function (ev) { if (ev.key === "Enter") { ev.preventDefault(); loadVoucher(); } });
  sendBtn.addEventListener("click", submit);
  renderPersons();
  loadTreatments();
  window.__sbwStarted = true;
})();
</script>
